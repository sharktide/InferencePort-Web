POST /v1/chat/completions
curl https://api.inferenceport.ai/v1/chat/completions \
  -H "Authorization: Bearer $IP_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "model": "deepseek-v4.1-flash-1017",
    "messages": [
      {"role": "user", "content": "Hello"}
    ]
  }'
POST /v1/images/generations
curl https://api.inferenceport.ai/v1/images/generations \
  -H "Authorization: Bearer $IP_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "model": "lightning-image-turbo",
    "prompt": "An architectural model on a desk"
  }'
POST /v1/embeddings
curl https://api.inferenceport.ai/v1/embeddings \
  -H "Authorization: Bearer $IP_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "model": "text-embedding-3-small",
    "input": "Make this searchable."
  }'
